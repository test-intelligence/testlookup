# ============================================================================
# TestLookup — container image manifest (SINGLE SOURCE OF TRUTH)
# ============================================================================
# Every surface that names a container image derives from this file:
#
#   scripts/release/offline-bundle.sh          which images get `docker save`d
#   scripts/release/import-bundle.sh           (via the bundle's images.list)
#   openshiftsetup/mirror-images.sh            INFRA_IMAGES to mirror
#   openshiftsetup/deploy-openshift-artifactory.sh
#   scripts/release/check_image_drift.py       asserts compose + k8s +
#                                              openshift-artifactory agree
#
# FORMAT — plain text, not YAML. Four whitespace-separated columns, `#`
# comments, blank lines ignored. Deliberately dependency-free so bash (awk /
# read), Python (stdlib only) and the CI drift guard can all parse it without
# PyYAML — the drift guard has to run in a job that installs nothing.
#
#   1. ref       full image reference incl. tag. App images use the literal
#                tag `@APP_TAG`, resolved at build time (bundle version /
#                build-YYYYMMDD-HHMMSS / BUILD_TAG_PLACEHOLDER depending on
#                the deploy path) — only their NAMES are drift-checked.
#   2. role      app   = built from this repo
#                infra = third-party, pulled from a public registry
#   3. surfaces  comma list of the deployment surfaces that reference it:
#                compose = docker-compose{,.release,.dev-lite}.yml
#                k8s     = k8s/base + k8s/overlays/**
#   4. bundle    core     = always in the offline bundle
#                llm      = only with `make offline-bundle ARGS=--with-llm`
#                external = a third-party PREREQUISITE the deployment needs
#                           but the bundle does not redistribute (licence —
#                           reason in the comment above the row). The
#                           operator mirrors it into the same registry under
#                           the same naming contract; mirror-images.sh and
#                           the OpenShift deploy script pull it from upstream
#                           on the operator's own connected host.
#                excluded = tracked for drift, never bundled (reason in the
#                           comment above the row)
#
# AIR-GAP NAMING CONTRACT — after import, every image lives under ONE registry:
#   infra:  <registry>/<ref>            e.g. myreg.corp/postgres:16-alpine
#   app:    <registry>/testlookup/<n>   e.g. myreg.corp/testlookup/backend:0.1.0
# This is exactly what k8s/overlays/openshift-artifactory/kustomization.yaml
# and openshiftsetup/mirror-images.sh already expect — do not change it
# without updating both.
#
# ref                                        role   surfaces      bundle
# ----------------------------------------------------------------------------

# ── Datastores ──────────────────────────────────────────────────────────────
# PostgreSQL License.
postgres:16-alpine                           infra  compose,k8s   core
# Valkey 8.1 (BSD-3-Clause), the Linux Foundation fork of Redis 7.2: same
# protocol for redis-py / Celery / kombu, so the `redis` service, the
# testlookup-redis k8s objects and every redis:// URL keep their names. It
# replaced redis:7-alpine, whose 7.4.x releases are RSALv2/SSPLv1.
valkey/valkey:8.1-alpine                     infra  compose,k8s   core
# MongoDB Community Server is SSPL-1.0 (not an OSI licence), so the offline
# bundle does not redistribute it: an EXTERNAL prerequisite the operator
# mirrors from upstream (see user-guide/air-gapped-install.md).
mongo:7                                      infra  compose,k8s   external

# ── Object storage ──────────────────────────────────────────────────────────
# versitygw (Apache-2.0): a single-binary S3 gateway, posix backend (buckets
# are directories, objects plain files under /data/s3 on the volume), health
# route /health. It replaced PGSTY Silo + mc (AGPL-3.0). Served from GHCR, so
# the name keeps its registry host: the air-gap path is
# <registry>/ghcr.io/versity/versitygw:<tag>. Buckets are created by the
# backend image's scripts/s3_setup.py (boto3) — no separate client image.
ghcr.io/versity/versitygw:v1.8.0             infra  compose,k8s   core

# ── Kubernetes-only helpers ─────────────────────────────────────────────────

# (k8s/base/worker-deployments.yaml). Also used by the offline-bundle CI job as
# the network-isolated checksum verifier.
busybox:1.36                                 infra  k8s           core

ollama/ollama:0.5.4                          infra  compose,k8s   llm
